#高斯消元
# 普通消元法
消元法是将方程组中的一方程的未知数用含有另一未知数的代数式表示，并将其带入到另一方程中，这就消去了一未知数，得到一解；或将方程组中的一方程倍乘某个常数加到另外一方程中去，也可达到消去一未知数的目的。消元法主要用于二元一次方程组的求解。

例如：
$$
\begin{cases}
4x + y=100 \\ 
x - y = 100 
\end{cases}
$$
那么，将，两式相加，消掉$y$得到，$5x = 200$即$x = 40$，代入式子中得到$y = -60$。

消元法理论的核心主要如下：
- 两方程互换，解不变；
- 一方程乘以非零数$k$解不变；
- 一方程乘以数$k$加上另一方程，解不变。
# 高斯消元法

其实高斯消元法在我们学习线性代数的时候经常用到，高斯消元法在将增广矩阵化为最简形后对于自由未知量的赋值，内容如下：
1. 增广矩阵行初等行变换为行最简形；
2. 还原线性方程组；
3. 求解第一个变量；
4. 补充自由未知量；
5. 列表示方程组通解。

例如：
对于方程组：
$$
\left\{
\begin{array}{ll}
2x_1+5x_3+6x_4 &=9 \\ 
x_3+x_4 &=-4 \\ 
2x_3+2x_4 &=-8 
\end{array} 
\right.
$$
转化为增广矩阵：
$$
{\Large
\begin{CD}
\left(
\begin{array}{cccc|c}
  2 & 0 & 5 & 6 & 9 \\
  0 & 0 & 1 & 1 & -4 \\
  0 & 0 & 2 & 2 & -8
\end{array}
\right)
@>r_3 - 2r_2>>
\left(
\begin{array}{cccc|c}
  2 & 0 & 5 & 6 & 9 \\
  0 & 0 & 1 & 1 & -4 \\
  0 & 0 & 0 & 0 & 0
\end{array}
\right)
\end{CD}
}
$$
转化为行阶梯型然后转化为最简型
$$
{\Large
\begin{CD}
@>\frac{r_1}{2}>>
\left(
\begin{array}{cccc|c}
  1 & 0 & \frac{5}{2} & 3 & \frac{9}{2} \\
  0 & 0 & 1 & 1 & -4 \\
  0 & 0 & 0 & 0 & 0
\end{array}
\right)
@>r_1-r_2 \times \frac{5}{2} >> 
\left(
\begin{array}{cccc|c}
  1 & 0 & 0 & \frac{1}{2} & \frac{29}{2} \\
  0 & 0 & 1 & 1 & -4 \\
  0 & 0 & 0 & 0 & 0
\end{array}
\right)
\end{CD}
}
$$
还原为线性方程组
$$
\left\{
\begin{array}{ll}
x_1 + \frac{1}{2}x_4 = \frac{29}{2} \\
x_3 + x_4 = -4
\end{array} 
\right.
$$
这不就是我们在线性代数中学的，用初等变换求解线性方程组吗。最后求解得：
$$
\left\{
\begin{array}{ll}
x_1 = -\frac{1}{2}x_4 + \frac{29}{2} \\
x_3 = -x_4 - 4
\end{array} 
\right.
$$
最后求解得
$$
\left\{
\begin{array}{ll}
x_1 = -\frac{1}{2}x_4 + \frac{29}{2} \\
x_2 = x_2 \\
x_3 = -x_4 - 4 \\
x_4 = x_4
\end{array} 
\right.
$$
而其中的$x_2$和$x_4$称为自由变量，其实就是任意值。

最后我们可以求出通解
$$
{\large
\begin{pmatrix}x_1 \\x_2 \\x_3 \\x_4\\\end{pmatrix} = \begin{pmatrix}0 \\1 \\0 \\0 \\\end{pmatrix} x_2 +\begin{pmatrix}-\frac{1}{2} \\0 \\-1 \\1 \\\end{pmatrix} x_4 +\begin{pmatrix}\frac{29}{2} \\0 \\-4 \\0 \\\end{pmatrix} = \begin{pmatrix}0 \\1 \\0\\0 \\\end{pmatrix} C_2 +\begin{pmatrix}-\frac{1}{2} \\0 \\-1 \\1 \\\end{pmatrix}C_4 + \begin{pmatrix}\frac{29}{2} \\0 \\-4 \\0 \\\end{pmatrix}}
$$
其中$C_2$和$C_4$为任意常数。
那我们接下来扩展到$n$个未知数的时候，也就是[题目1]([P3389 【模板】高斯消元法 - 洛谷](https://www.luogu.com.cn/problem/P3389))和[题目二]([P2455 [SDOI2006] 线性方程组 - 洛谷](https://www.luogu.com.cn/problem/P2455))。
其实就是让我们求
$$
\begin{cases} a_{1,1}x_1 + a_{1,2}x_2 + \cdots + a_{1,n}x_n &= b_1 \\ a_{2,1}x_1 + a_{2,2}x_2 + \cdots + a_{2,n}x_n &= b_2 \\ &\cdots \\ a_{n,1}x_1 + a_{n,2}x_2 + \cdots + a_{n,n}x_n &= b_n \end{cases}
$$
思路跟上面说的一样，就是让我们转化为行阶梯型的矩阵，然后求解。
此处展示题目二的代码：
```cpp
#include <bits/stdc++.h>
#define ull unsigned long long
#define i128 __int128_t
using namespace std;
const int INF = 1e18;
const int N = 105;
const int MOD = 1e9 + 7;
const double eqs = 1e-8;
int n;
double matrix[N][N], ans[N];
int main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);
	cout << fixed << setprecision(4);
	cin >> n;
	for (int i = 1; i <= n; ++i) 
		for (int j = 1; j <= n + 1; ++j) 
			cin >> matrix[i][j];
	int r = 1; // 在这里，我们去寻找每一列中的主元
	for (int c = 1; c <= n; ++c) {
		int p = r;
		// 去寻找尽量大的数当作主元
		for (int i = r; i <= n; ++i) 
			if (fabs(matrix[p][c]) < fabs(matrix[i][c])) p = i;
		// 当前列没有有效的主元（即全为零）
		// 那么就跳过这一列的主元（不跳过行，所以需要另外再创建一个r的参数作为寻找过的行）
		if (fabs(matrix[p][c]) <= eqs) continue;
		// 设置主元
		swap(matrix[r], matrix[p]);
		// 消去当前列下方的所有元素
		for (int i = r + 1; i <= n; ++i) {
			double fac = matrix[i][c] / matrix[r][c];
			for (int j = 1; j <= n + 1; ++j) matrix[i][j] -= fac * matrix[r][j];
		}
		r++; // 非零行加1
	}
	for (int i = r; i <= n; ++i) {
		bool flag = true;
		for (int j = 1; j <= n; ++j) {
			if (fabs(matrix[i][j]) > eqs) {
				flag = false;
				break;
			}
		}
		// 系数全零，但是右端项非零，形成了0 = c的式子，所以无解
		if (flag && fabs(matrix[i][n + 1]) > eqs) {
			cout << -1 << '\n';
			return 0;
		}
	}
	// 倘若存在0 = 0的式子，那么就存在无穷多解
	if (r <= n) {
		cout << 0 << '\n';
		return 0;
	}
	for (int i = n; i >= 1; --i) {
		// 开始消元
		for (int j = i + 1; j <= n; ++j) matrix[i][n + 1] -= ans[j] * matrix[i][j];
		// 求解出对应x
		ans[i] = matrix[i][n + 1] / matrix[i][i];
	}
	for (int i = 1; i <= n; ++i) cout << 'x' << i << '=' << ans[i] << '\n';
	return 0;
}
```

# 求行列式

当然我们也可以用高斯消元来求行列式的值，[题目](https://www.luogu.com.cn/problem/P7112)。
代码如下：
```cpp
#include <bits/stdc++.h>
#define int long long
#define ull unsigned long long
#define i128 __int128_t
using namespace std;
const int INF = 1e18;
const int N = 605;
const int MOD = 1e9 + 7;
int n,p,ans = 1,sign = 1;
int matrix[N][N];
signed main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);
	cout << fixed << setprecision(15);
	cin >> n >> p;
	for (int i = 1; i <= n; ++i) for (int j = 1; j <= n; ++j) {
		cin >> matrix[i][j];
		matrix[i][j] %= p;
	}
	for (int i = 1; i <= n; ++i) {
		for (int j = i + 1; j <= n; ++j) {
			while(matrix[j][i]) {
				// 这里我们消元为0的过程不能用普通的高斯消元，原因在于模数p可能为非质数
				// 于是我们可以借鉴辗转相处的过程，通过欧几里得算法将matrix[j][i]的值转化为0
				int t = matrix[i][i] / matrix[j][i];
				for (int k = i;k <= n; ++k) 
					matrix[i][k] = (matrix[i][k] - t * matrix[j][k] % p + p) % p;
				swap(matrix[i],matrix[j]);
				sign = -sign;
			}
		}
	}
	for (int i=1;i<=n;++i) ans = ans * matrix[i][i] % p;
	ans = (ans * sign % p + p) % p;
	cout<<ans<<'\n';
	return 0;
}
```

# 矩阵求逆

当然我们也可以通过高斯消元解决逆矩阵的问题。[【模板】矩阵求逆](https://www.luogu.com.cn/problem/P4783)。
代码如下，不再说明。
```cpp
#include <bits/stdc++.h>
#define int long long
#define ull unsigned long long
#define i128 __int128_t
using namespace std;
const int INF = 1e18;
const int N = 405;
const int MOD = 1e9 + 7;
int n;
int a[N][N],ans[N][N];
int qpow(int base, int power) {
	int res = 1;
	while (power) {
		if (power & 1) res = base * res % MOD;
		base = base * base % MOD;
		power >>= 1;
	}
	return res;
}
signed main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);
	cout << fixed << setprecision(15);
	cin >> n;
	for (int i = 1; i <= n; ++i) {
		ans[i][i] = 1;
		for (int j = 1; j <= n; ++j) cin >> a[i][j];
	}
	for (int i = 1; i <= n; ++i) {
		int p = i;
		for (int j = i + 1; j <= n; ++j) if (a[p][i] < a[j][i]) p = j;
		if (a[p][i] == 0) {
			cout << "No Solution\n";
			return 0;
		}
		swap(a[p], a[i]);
		swap(ans[p], ans[i]);
		int inv = qpow(a[i][i], MOD - 2);
		for (int j = i + 1; j <= n; ++j) {
			int d = a[j][i] * inv % MOD;
			for (int k = 1; k <= n; ++k) {
				a[j][k] = (a[j][k] - d * a[i][k] % MOD + MOD) % MOD;
				ans[j][k] = (ans[j][k] - d * ans[i][k] % MOD + MOD) % MOD;
			}
		}
	}
	for (int i = n; i >= 1; --i) {
		int inv = qpow(a[i][i], MOD - 2);
		for (int j = i - 1; j >= 1; --j) {
			int d = a[j][i] * inv % MOD;
			for (int k = 1; k <= n; ++k) {
				a[j][k] = (a[j][k] - d * a[i][k] % MOD + MOD) % MOD;
				ans[j][k] = (ans[j][k] - d * ans[i][k] % MOD + MOD) % MOD;
			}
		}
		a[i][i] = a[i][i] * inv % MOD;
		for (int j = 1; j <= n; ++j) ans[i][j] = ans[i][j] * inv % MOD;
	}
	for (int i = 1; i <= n; ++i) {
		for (int j = 1; j <= n; ++j) cout << ans[i][j] << ' ';
		cout << '\n';
	}
	return 0;
}
```

# 高斯消元法求解异或方程组

异或方程组是指形如
$$
\begin{cases}
a_{1,1}x_1 \oplus a_{1,2}x_2 \oplus \cdots \oplus a_{1,n}x_n &= b_1\\
a_{2,1}x_1 \oplus a_{2,2}x_2 \oplus \cdots \oplus a_{2,n}x_n &= b_2\\
\cdots &\cdots \\ a_{m,1}x_1 \oplus a_{m,2}x_2 \oplus \cdots \oplus a_{m,n}x_n &= b_m \\
\end{cases}
$$
的方程组，其中 $\oplus$ 表示「按位异或」（即 `xor` 或 C++ 中的 `^`），且式中所有系数/常数（即 $a_{i,j}$ 与 $b_i$）均为 $0$ 或 $1$．

由于「异或」符合交换律与结合律，故可以按照高斯消元法逐步消元求解．值得注意的是，我们在消元的时候应使用「异或消元」而非「加减消元」，且不需要进行乘除改变系数（因为系数均为 $0$ 和 $1$）。

所以我们可以将以上的异或方程组构造为如下增广矩阵：
$$
\left (
\begin{array}{ccccc|c}
a_{1,1} & a_{1,2} & a_{1,3} & \cdots & a_{1,m} & b_1 \\
a_{2,1} & a_{2,2} & a_{2,3} & \cdots & a_{2,m} & b_2 \\
\vdots & \vdots & \vdots & \ddots & \vdots & \vdots \\
a_{n,1} & a_{n,2} & a_{n,3} & \cdots & a_{n,m} & b_n
\end{array}
\right )
$$
其中一共有$m$个未知数，$n$个方程式，其中任意一个$a_{i,j}$为$0$或$1$，通过异或运算即可转化为最简行阶梯型的矩阵。

注意到异或方程组的增广矩阵是 $01$ 矩阵（矩阵中仅含有 $0$ 与 $1$），所以我们可以使用 C++ 中的 `std::bitset` 进行优化，将时间复杂度降为 $O(\dfrac{n^2m}{\omega})$，其中 $n$ 为元的个数，$m$ 为方程条数，$\omega$ 一般为 $32$（与机器有关）。



例题为：
![](学习笔记/Algorithm/已经做过的比赛题目/2025年北京市赛/2025北京市大学生程序设计竞赛暨“小米杯”全国邀请赛.pdf#page=10)

在这一题中，题目要求我们求出矩阵$B$的值，那么存在关键线索$A \times B_{:j} = C_{:j}$，即$C$中的某一列就是由$B$中的某一列与$A$相乘得到的，那么把$B$中的那一列当作未知数就可以作高斯消元了，然后由于每次高斯消元的对象都是整个$A$，那么我们可以一次性处理所有的$C$列，然后求出$B$的列，时间复杂度为$O(p*n^2m)$，需要使用`bitset`优化，时间复杂度为$O(\frac{n^2(m+p)}{w})$。

代码如下：
```cpp
#include <bits/stdc++.h>
#define ull unsigned long long
#define i128 __int128_t
using namespace std;
const int N = 1005;
const int MOD = 1e9 + 7;
int n,m,p,r = 1;
bitset<N> a[N],c[N];
int pivot[N];
bool ans[N][N];
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    cout << fixed << setprecision(15);
    cin >> n >> m >> p;
    for (int i = 1; i <= n; ++i) for (int j = 1; j <= m; ++j) {
        int x;
        cin >> x;
        if (x) a[i].set(j);
    }
    for (int i = 1; i <= n; ++i) for (int j = 1; j <= p; ++j) {
        int x;
        cin >> x;
        if (x) c[i].set(j);
    }
    for (int i = 1; i <= m; ++i) {
        int p = r;
        for (int j = r; j <= n; ++j) if (a[j].test(i)) {
            p = j;
            break;
        }
        if (!a[p].test(i)) continue;
        swap(a[r], a[p]);
        swap(c[r], c[p]);
        for (int j = 1; j <= n; ++j) if (j != r && a[j].test(i)) {
            a[j] ^= a[r];
            c[j] ^= c[r];
        }
        pivot[r] = i;
        r++;
    }
    for (int i = r; i <= n; ++i) {
        if (c[i].count()) {
            cout << "No\n";
            return 0;
        }
    }
    for (int i=1;i<r;++i) {
        int var = pivot[i];
        for (int j=1;j<=p;++j) ans[var][j] = c[i][j];
    }
    cout << "Yes\n";
    for (int i = 1; i <= m; ++i) {
        for (int j = 1; j <= p; ++j) {
            if (ans[i][j]) cout << 1 << ' ';
            else cout << 0 << ' ';
        }
        cout << '\n';
    }
    return 0;
}
```
