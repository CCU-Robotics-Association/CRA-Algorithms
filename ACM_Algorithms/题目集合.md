[A - GD 终极节奏实验室](https://codeforces.com/gym/105924/problem/A)
最优解 #单调栈
```c++
#include <bits/stdc++.h>
#define int long long
#define ull unsigned long long
#define i128 __int128_t
using namespace std;
const int INF = 1e18;
const int N = 1e6 + 5;
const int MOD = 1e9 + 7;
void solve() {
	int n,ans = 0;
	cin>>n;
	vector<int> a(n + 1),l(n + 1),r(n + 1),temp;
	for (int i=1;i<=n;++i) cin>>a[i];
	for (int i=1;i<=n;++i) {
		if (temp.empty()) temp.emplace_back(i);
		else {
			while(!temp.empty() && a[i] % a[temp.back()]) {
				r[temp.back()] = i - 1;
				temp.pop_back();
			}
			temp.emplace_back(i);
		}
	}
	while(!temp.empty()) {
		r[temp.back()] = n;
		temp.pop_back();
	}
	for (int i=n;i >= 1;--i) {
		if (temp.empty()) temp.emplace_back(i);
		else {
			while(!temp.empty() && (a[i] % a[temp.back()] || a[i] == a[temp.back()])) {
				l[temp.back()] = i + 1;
				temp.pop_back();
			}
			temp.emplace_back(i);
		}
	}
	while(!temp.empty()) {
		l[temp.back()] = 1;
		temp.pop_back();
	}
	for (int i=1;i<=n;++i) ans += (r[i] - i + 1) * (i - l[i] + 1);
	cout<<ans<<'\n';
}
signed main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);
	cout << fixed << setprecision(15);
	int t = 1;
	// cin >> t;
	while (t--) {
		solve();
	}
	return 0;
}
```

[H - 王国------迁移](https://codeforces.com/gym/105924/problem/H)
#二分 #Hall定理
```C++
#include <bits/stdc++.h>
#define int long long
#define ull unsigned long long
#define i128 __int128_t
using namespace std;
const int INF = 1e18;
const int N = 1e6 + 5;
const int MOD = 1e9 + 7;
void solve() {
    int n, sumb = 0, ans = 0;
    cin >> n;
    vector<int> a(n + 1), b(n + 1), c(n + 1);
    for (int i = 1; i <= n; ++i) cin >> a[i];
    for (int i = 1; i <= n; ++i) cin >> b[i], sumb += b[i];
    for (int i = 1; i <= n; ++i) cin >> c[i];
    int l = 1, r = 1e12;
    auto check = [&](int mid)->bool {
        int sum = 0;
        vector<int> cnt(n + 1);
        for (int i = 1; i <= n; ++i) cnt[i] = mid / c[i], sum += cnt[i];
        if (sum < sumb) return false;
        else {
            for (int i = 1; i <= n; ++i) if (b[i] > sum - cnt[a[i]]) return false;
            return true;
        }
        };
    while (l <= r) {
        int mid = (l + r) >> 1;
        if (check(mid)) {
            ans = mid;
            r = mid - 1;
        }
        else l = mid + 1;
    }
    cout << ans << '\n';
}
signed main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    cout << fixed << setprecision(15);
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```
