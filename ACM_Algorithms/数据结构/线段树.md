#线段树
线段树是算法竞赛中常用的用来维护**区间信息**的数据结构

线段树的时间复杂度为$O(\log N)$，在这个时间复杂度内可以实现**单点修改、区间修改、区间查询（区间求和，求区间最大值，求区间最小值**）等操作，常常被用来做RMQ等区间问询的问题。

# 上模板
不带mod余数版本

```cpp
template <typename T>
class RangeTree {
    vector<T> tree, add_lazy, mul_lazy;
    int n, root, end, length;
    void build(int s, int t, int p, const vector<T>& a) {
        if (s == t) {
            tree[p] = a[s];
            return;
        }
        int mid = ((s + t) >> 1);
        build(s, mid, p * 2, a);
        build(mid + 1, t, p * 2 + 1, a);
        tree[p] = tree[p * 2] + tree[p * 2 + 1];
    }
    void maintain(int l, int r, int p) {
        if (mul_lazy[p] == 1 && add_lazy[p] == 0 || l == r) return;
        int mid = ((l + r) >> 1), left = p * 2, right = p * 2 + 1;
        tree[left] = tree[left] * mul_lazy[p] + add_lazy[p] * (mid - l + 1), mul_lazy[left] *= mul_lazy[p], add_lazy[left] = add_lazy[left] * mul_lazy[p] + add_lazy[p];
        tree[right] = tree[right] * mul_lazy[p] + add_lazy[p] * (r - mid), mul_lazy[right] *= mul_lazy[p], add_lazy[right] = add_lazy[right] * mul_lazy[p] + add_lazy[p];
        mul_lazy[p] = 1, add_lazy[p] = 0;
    }
    T range_query(int l, int r, int s, int t, int p) {
        if (l <= s && t <= r) return tree[p];
        int mid = ((s + t) >> 1), sum = 0;
        maintain(s, t, p);
        if (l <= mid) sum += range_query(l, r, s, mid, p * 2);
        if (r > mid) sum += range_query(l, r, mid + 1, t, p * 2 + 1);
        return sum;
    }
    void update_add(int l, int r, int s, int t, int p, T val) {
        if (l <= s && t <= r) {
            tree[p] += (t - s + 1) * val, add_lazy[p] += val;
            return;
        }
        int mid = ((s + t) >> 1);
        maintain(s, t, p);
        if (l <= mid) update_add(l, r, s, mid, p * 2, val);
        if (r > mid) update_add(l, r, mid + 1, t, p * 2 + 1, val);
        tree[p] = tree[p * 2] + tree[p * 2 + 1];
    }
    void update_mul(int l, int r, int s, int t, int p, T val) {
        if (l <= s && t <= r){
            tree[p] *= val, mul_lazy[p] *= val, add_lazy[p] *= val;
            return;
        }
        int mid = ((s + t) >> 1);
        maintain(s, t, p);
        if (l <= mid) update_mul(l, r, s, mid, p * 2, val);
        if (r > mid) update_mul(l, r, mid + 1, t, p * 2 + 1, val);
        tree[p] = tree[p * 2] + tree[p * 2 + 1];
    }
public:
    explicit RangeTree<T>(vector<T>& a) {
        n = a.size();
        length = n * 4;
        tree = vector<T>(length, 0);
        add_lazy = vector<T>(length, 0);
        mul_lazy = vector<T>(length, 1);
        end = n - 1;
        root = 1;
        build(0, end, 1, a);
    }
    void range_add(int l, int r, T val) { return update_add(l - 1, r - 1, 0, end, root, val); };
    void range_mul(int l, int r, T val) { return update_mul(l - 1, r - 1, 0, end, root, val); };
    T range_sum(int l, int r) { return range_query(l - 1, r - 1, 0, end, root); };
    void show(int p, int depth = 0) {
        if (p > length || tree[p] == 0) return;
        show(p * 2, depth + 1);
        for (int i = 0; i < depth; ++i) putchar('\t');
        printf("%d:%d\n", tree[p], add_lazy[p]);
        show(p * 2 + 1, depth + 1);
    }
};
```

```cpp
    int n, root, end, length, MOD = LONG_LONG_MAX - 1;
    void build(int s, int t, int p, const vector<T>& a) {
        if (s == t) {
            tree[p] = a[s];
            return;
        }
        int mid = ((s + t) >> 1);

        int mid = ((l + r) >> 1), left = p * 2, right = p * 2 + 1;
        tree[left] = (tree[left] * mul_lazy[p] % MOD + add_lazy[p] * (mid - l + 1) % MOD) % MOD, mul_lazy[left] = mul_lazy[p] * mul_lazy[left] % MOD, add_lazy[left] = (add_lazy[left] * mul_lazy[p] % MOD + add_lazy[p]) % MOD;
        tree[right] = (tree[right] * mul_lazy[p] % MOD + add_lazy[p] * (r - mid) % MOD) % MOD, mul_lazy[right] = mul_lazy[p] * mul_lazy[right] % MOD, add_lazy[right] = (add_lazy[right] * mul_lazy[p] % MOD + add_lazy[p]) % MOD;
        mul_lazy[p] = 1, add_lazy[p] = 0;
    }

        if (l <= s && t <= r) return tree[p];

        maintain(s, t, p);
        if (l <= mid) sum = (sum + range_query(l, r, s, mid, p * 2)) % MOD;
        if (r > mid) sum = (sum + range_query(l, r, mid + 1, t, p * 2 + 1)) % MOD;
        return sum;
    }
    void update_add(int l, int r, int s, int t, int p, T val) {
        if (l <= s && t <= r) {
            tree[p] = (tree[p] + (t - s + 1) * val) % MOD, add_lazy[p] = (val + add_lazy[p]) % MOD;
            return;
        }
        int mid = ((s + t) >> 1);
        maintain(s, t, p);
        if (l <= mid) update_add(l, r, s, mid, p * 2, val);
        if (r > mid) update_add(l, r, mid + 1, t, p * 2 + 1, val);
        tree[p] = (tree[p * 2] + tree[p * 2 + 1]) % MOD;
    }
    void update_mul(int l, int r, int s, int t, int p, T val) {
        if (l <= s && t <= r){
            tree[p] = val * tree[p] % MOD, mul_lazy[p] = val * mul_lazy[p] % MOD, add_lazy[p] = val * add_lazy[p] % MOD;
            return;
        }
        int mid = ((s + t) >> 1);
        maintain(s, t, p);
        if (l <= mid) update_mul(l, r, s, mid, p * 2, val);
        if (r > mid) update_mul(l, r, mid + 1, t, p * 2 + 1, val);
        tree[p] = (tree[p * 2] + tree[p * 2 + 1]) % MOD;
    }
public:
    explicit RangeTreeWithMOD<T>(vector<T>& a) {

        add_lazy = vector<T>(length, 0);
        mul_lazy = vector<T>(length, 1);
        end = n - 1;
        root = 1;
        build(0, end, 1, a);
    }
    explicit RangeTreeWithMOD<T>(vector<T>& a, int MOD) {

        tree = vector<T>(length, 0);
        add_lazy = vector<T>(length, 0);
        mul_lazy = vector<T>(length, 1);

    }

    }
```

# 简介
## 建树
线段树本质上是一棵二叉树，对于一个线段用二叉树表示。

![](资源归档/assets/103.png)

那么我们应该如何实现呢。

如果你表示线段的和，那么最上面的根节点的权值表示的是这个线段 1 ∼ 4 的和。根的两个儿子分别表示这个线段中 1 ∼ 2 的和，与 3 ∼ 4 的和。以此类推。

有个大小为$5$的数组$a = \left\{10, 11, 12 ,13 ,14\right\}$，要将其转化为线段树，有以下做法：设线段树的根节点编号为$1$，用数组$d$来保存我们的线段树，$d_{i}$用来保存线段树上编号为$i$的节点的值（这里每个节点所维护的值就是这个节点所表示的区间总和）。
建图如下：

![](资源归档/assets/104.svg)

图中每个节点中用红色字体标明的区间，表示该节点管辖的$a$数组上的位置区间。如$d_{1}$所管辖的区间就是$[1,5] (a_{1},a_{2},...,a_{5})$，即$d_{1}$所保存的值是$a_{1}+a_{2}+...+a_{5}$，$d_{1} = 60$表示的是$a_{1}+a_{2}+...+a_{5} = 60$。

通过观察不难发现，$d_{i}$的左儿子节点就是$d_{i\times2}$，$d_{i}$的右儿子节点就是$d_{i\times2+1}$。如果$d_{i}$表示的是区间$[s,t]$（即$d_{i} = a_{s} + a_{s+1}+...+a_{t}$）的话，那么$d_{i}$的左儿子节点表示的是区间$[s,\frac{s+t}{2}]$，$d_{i}$的右儿子表示的是区间$[\frac{s+t}{2}+1,t]$。

在实现时，我们考虑递归建树。设当前的根节点为$p$，如果根节点管辖的区间长度已经是$1$，则可以直接根据$a$数组上相应位置的值初始化该节点。否则我们将该区间从中点处分割为两个子区间，分别进入左右子节点递归建树，最后合并两个子节点的信息。

代码：

```cpp
//对[s,t]区间递归建树，当前节点的标号为p，a为需要建树的数组
void build(int s, int t, int p, const vector<T>& a) {
    //倘若定位到某个点上，则直接将该节点接入tree数组上
    if (s == t) {
        tree[p] = a[s];
        return;
    }
    //取区间中间
    int mid = ((s + t) >> 1);
    //中间左边建树
    build(s, mid, p * 2, a);
    //中间右边建树
    build(mid + 1, t, p * 2 + 1, a);
    //将左边和右边的树结合
    //这个操作是建树中比较重要的操作，名叫pushup
    tree[p] = tree[p * 2] + tree[p * 2 + 1];
}
```

关于线段树的空间：

线段树通常采用堆式存储（$2p$是$p$的左儿子，$2p+1$是$p$的右儿子），若有$n$个叶子节点，则$d$数组的最大范围为$2^{\left \lceil \log n \right \rceil +1}$。一般来说我们通常建树的时候将数组的大小建为$4n$即可。
## 区间查询
区间查询，比如区间$[l,r]$的总和（即$a_{l}+a_{l+1}+...+a_{r}$）

![](资源归档/assets/105.svg)

仍然以最开始的图为例，如果要查询区间$[1,5]$的和，那直接获取$d_{1}$的值（60）即可。

如果要查询的区间为$[3,5]$，此时就不能直接获取区间的值，但是$[3,5]$可以拆成$[3]$和$[4,5]$，可以通过合并这两个区间的答案来求得这个区间的答案。

一般地，如果要查询的区间是$[l,r]$，则可以将其拆成最多为$O(\log n)$个 **极大** 的区间，合并这些区间即可求出$[l,r]$的答案。

代码：

```cpp
//递归查询区间[l,r]为我们需要查询的区间，[s,t]为我们目前查找到的区间，p为当前区间节点编号
T RangeCheck(int l, int r, int s, int t, int p) {
    //我们目前查找到的区间位于需要查找的区间之内，那么返回当前区间的值
    if (l <= s && t <= r) return tree[p];
    //取中间
    int mid = ((s + t) >> 1), sum = 0;
    //数组维护，下面会讲到
    maintain(s, t, p);
    //下面的情况为我们查找到的数组并非在目标数组内
    //倘若查找到的区间的中间位置大于目标区间左边，那么就往左边找，如果中间位置小于l，
    //说明查找到的中间左边的区间不在目标区间内，那么就不找
    if (l <= mid) sum += RangeCheck(l, r, s, mid, p * 2);
    //往右边的同理倘若目标的右边界大于中间，那么找到的右半部分为目标区间之内，如果不是则反之
    if (r > mid) sum += RangeCheck(l, r, mid + 1, t, p * 2 + 1);
    return sum;
}
```

## 区间修改
对一个$[l,r]$的区间进行修改，比如加一个值或者乘多少倍，就需要对数组进行区间修改

### 加一个值
对一个区间加上某个值，把所有$[l,r]$区间中的所有节点都遍历一遍、修改一遍，时间复杂度无法承受。所以这里我们需要引入一个叫做**懒惰标记**的东西。

**懒惰标记**：

就是通过延迟对节点信息的更改，从而减少可能不必要的操作次数。每次执行修改时，我们通过打标记的方法表明该节点对应的区间在某一次操作中被更改，但不更新该节点的子节点的信息。实质性的修改则在下一次访问带有标记的节点时才进行。

具体实现是通过先递归寻找目标的范围，将范围对应的$d_{i}$进行修改并且进行记录，表示下面所有的子节点都要进行相加。还是以前面的数组举例，现在我们准备给$[3,5]$的范围内的每个数字都加上$5$。根据区间查询，我们就可以查询到$[3]$和$[4,5]$。给对应的分段上打上记号。

![](资源归档/assets/106.svg)

代码如下：

```cpp
//l，r代表目标区间，s，t代表查找到的区间，p为区间节点编号，val为加的值
void update_add(int l, int r, int s, int t, int p, T val) {
    //与上面一样，倘若查找到的区间在目标区间内则进行操作
    if (l <= s && t <= r) {
        //对该编号加上叶子节点累加的值，并且该编号的懒节点加上对应的值
        tree[p] += (t - s + 1) * val, add_lazy[p] += val;
        return;
    }
    //查找到的中间的值
    int mid = ((s + t) >> 1);
    //维护区间数组，也是维护懒节点数组
    maintain(s, t, p);
    //说明查找到区间的左边为目标区间内，则再对左边进行更新
    if (l <= mid) update_add(l, r, s, mid, p * 2, val);
    //说明查找到区间的右边为目标区间内，再对右边进行更新
    if (r > mid) update_add(l, r, mid + 1, t, p * 2 + 1, val);
    //该编号节点加上左边部分节点和右边部分节点的值
    tree[p] = tree[p * 2] + tree[p * 2 + 1];
}
```

对上面的代码来说我们都看到一个叫`maintain`的方法，该方法的作用是维护区间数组和懒节点数组，`maintain`方法代码如下：

```cpp
//l，r为需要维护的节点的左边界和右边界，p为该区间节点的编号
//数组中mul_lazy是为了下文区间乘法修改做准备
void maintain(int l, int r, int p) {
    //倘若该区间没有任何修改，那么直接返回
    if (mul_lazy[p] == 1 && add_lazy[p] == 0 || l == r) return;
    //查找区间的中间位置，left为左区间节点编号，right为右区间节点编号
    int mid = ((l + r) >> 1), left = p * 2, right = p * 2 + 1;
    //区间左边节点的修改，和左区间懒节点更新和
    tree[left] = tree[left] * mul_lazy[p] + add_lazy[p] * (mid - l + 1), mul_lazy[left] *= mul_lazy[p], add_lazy[left] = add_lazy[left] * mul_lazy[p] + add_lazy[p];
    //区间右边节点的修改，和右区间懒节点更新
    tree[right] = tree[right] * mul_lazy[p] + add_lazy[p] * (r - mid), mul_lazy[right] *= mul_lazy[p], add_lazy[right] = add_lazy[right] * mul_lazy[p] + add_lazy[p];
    //当前懒节点重置
    mul_lazy[p] = 1, add_lazy[p] = 0;
}
```

# 动态开点线段树
原先静态线段树，是直接创建长度为$4n$的数组，而动态的实际上是只有用到的时候才创建节点。
静态线段树适合：
- 数组长度已知
- 下标范围不大
- 整棵树都会用到
而动态开点线段树适合：
- 下边范围大，比如`[1,1e9]`、`[1,1e18]`
- 但是实际只有很少一部分区间被修改
- 省内存

## 开点过程
动态线段树的主要过程如下。
开始时只有一个节点，存储一个范围
![[Pasted image 20260624210814.png]]

当我们给`[1,8]`范围内的元素添加或减少某个值时，发现范围并没有被创建，于是就会创建对应范围的节点。
![[Pasted image 20260624211802.png]]
所以我们代码的实现也就体现在如此。当我们需要某个区间时，动态的开点去描述这个区间。代码如下。

```cpp
class DynamicSegTree {
private:
	// 每个节点的初始化
	struct node {
		int lc, rc, sum, add, mul;
		node() : lc(0), rc(0), sum(0), add(0), mul(1) {}
	};
	vector<node> tree;
	int L, R, root;
	int newNode() {
		tree.push_back(node());
		return tree.size() - 1;
	}
	// 跟静态的一样，pushup节点
	void pushup(int p) { 
	tree[p].sum = 
	(tree[p].lc ? tree[tree[p].lc].sum : 0) + 
	(tree[p].rc ? tree[tree[p].rc].sum : 0); 
	}
	// 这里多加了个apply方法，是为了让模板代码更加简介
	void apply(int p, int mul, int add, int len) {
		tree[p].sum = tree[p].sum * mul + add * len;
		tree[p].mul *= mul;
		tree[p].add = tree[p].add * mul + add;
	}
	// 这里对应上面模板的maintain方法（网上说应该叫pushdown方法才对）
	void pushdown(int p,int l,int r) {
		if (l == r) return;
		if (tree[p].add == 0 && tree[p].mul == 1) return;
		int mid = (l + r) >> 1;
		if (!tree[p].lc) tree[p].lc = newNode();
		if (!tree[p].rc) tree[p].rc = newNode();
		apply(tree[p].lc, tree[p].mul, tree[p].add, mid - l + 1);
		apply(tree[p].rc, tree[p].mul, tree[p].add, r - mid);
		tree[p].add = 0;
		tree[p].mul = 1;
	}
	void range_add(int& p, int l, int r, int ql, int qr, int val) {
		if (!p) p = newNode();
		if (ql <= l && qr >= r) {
			apply(p, 1, val, r - l + 1);
			return;
		}
		pushdown(p, l, r);
		int mid = (l + r) >> 1;
		if (mid >= ql) range_add(tree[p].lc, l, mid, ql, qr, val);
		if (qr > mid) range_add(tree[p].rc, mid + 1, r, ql, qr, val);
		pushup(p);
	}
	void range_mul(int& p, int l, int r, int ql, int qr, int val) {
		if (!p) p = newNode();
		if (ql <= l && qr >= r) {
			apply(p, val, 0, r - l + 1);
			return;
		}
		pushdown(p, l, r);
		int mid = (l + r) >> 1;
		if (mid >= ql) range_mul(tree[p].lc, l, mid, ql, qr, val);
		if (mid < qr) range_mul(tree[p].rc, mid + 1, r, ql, qr, val);
		pushup(p);
	}
	int query(int p, int l, int r, int ql, int qr) {
		if (!p) return 0;
		if (ql <= l && qr >= r) return tree[p].sum;
		pushdown(p, l, r);
		int mid = (l + r) >> 1, ans = 0;
		if (mid >= ql) ans += query(tree[p].lc, l, mid, ql, qr);
		if (mid < qr) ans += query(tree[p].rc, mid + 1, r, ql, qr);
		return ans;
	}
public:
	DynamicSegTree(int L, int R) : L(L), R(R) {
		tree.push_back(node());
		root = newNode();
	}
	void add(int l, int r, int val) { range_add(root, L, R, l, r, val); }
	void mul(int l, int r, int val) { range_mul(root, L, R, l, r, val); }
	int sum(int l, int r) { return query(root, L, R, l, r); }
};
```
