#最小生成树
# 洛谷P3366 【模板】最小生成树
## 题目描述
如题，给出一个无向图，求出最小生成树，如果该图不连通，则输出 `orz`。

## 输入格式
第一行包含两个整数 $N,M$，表示该图共有 $N$ 个结点和 $M$ 条无向边。

接下来 $M$ 行每行包含三个整数 $X_i,Y_i,Z_i$，表示有一条长度为 $Z_i$ 的无向边连接结点 $X_i,Y_i$。

## 输出格式
如果该图连通，则输出一个整数表示最小生成树的各边的长度之和。如果该图不连通则输出 `orz`。

## 输入输出样例 #1
### 输入 #1
```plain
4 5
1 2 2
1 3 2
1 4 3
2 3 4
3 4 3
```

### 输出 #1
```plain
7
```

## 说明/提示
数据规模：

对于 $20\%$ 的数据，$N\le 5$，$M\le 20$。

对于 $40\%$ 的数据，$N\le 50$，$M\le 2500$。

对于 $70\%$ 的数据，$N\le 500$，$M\le 10^4$。

对于 $100\%$ 的数据：$1\le N\le 5000$，$1\le M\le 2\times 10^5$，$1\le Z_i \le 10^4$。

# Prime算法（普里姆算法）
## 定义
该算法在查找最小生成树时，将顶点分为两类：

+ 一类为在查找过程中已经包含在生成树中的顶点（A类）
+ 第二类为未进入最小生成树的点（B类）

## 过程
对于给定的连通图，起始状态全部顶点都是未进入生成树的点（B类）。

1. 在寻找最小的生成树的时候，选定任意一个顶点作为起始点，并将之从未进入生成树的点转到进入生成树的点。
2. 然后找到B类中到A类中的顶点之间的最小权值的点，将它从B类移入A类，移入A类时经过的边即是最小生成树的边之一。
3. 最后重复2的过程直到每一个点都进入最小生成树。

## 第一次写的代码：
```cpp
#include <bits/stdc++.h>
#define ll long long
using namespace std;
int n,m,ans = 0;
//dis[i][j]存储的是从i到j的权值
int dis[5010][5010];
//via数组的存储的是via[i]的i是否进入最小生成树
bool via[5010];
//g数组存储的是边的集合
vector<vector<int>> g(5010,vector<int> ());
//a数组存储的是进入最小生成树的节点的集合
vector<int> a;
void prime(){
	a.push_back(1);
	via[1] = true;
    //判断是否还有元素未进入已加入最小生成树的数组中，如果有则继续循环，如果没有就退出循环
	while(a.size() != n){
        //minn变量存储的是已经进入最小的生成树的节点与未进入最小生成树的节点连通的权值的最小值
        //f变量的作用是存储新进入最小生成树节点的编号
		int minn = 1e5,f;
        //循环进入最小生成树的节点
		for (int i=0;i<a.size();++i){
            //循环进入最小生成树的节点的边
			for (int j=0;j<g[a[i]].size();++j){
                //如果该节点未进入最小生成树
				if (!via[g[a[i]][j]]){
                    //判断是否为最小节点
					if (minn > dis[a[i]][g[a[i]][j]]){
						minn = dis[a[i]][g[a[i]][j]];
						f = g[a[i]][j];
					}
				}
			}
		}
		ans+=minn;
		a.push_back(f);
		via[f]=true;
	}
}
void solve() {
	cin>>n>>m;
	for (int i=0;i<m;++i){
		int x,y,z;
		cin>>x>>y>>z;
		if (dis[x][y] == 0){
			g[x].push_back(y);
			g[y].push_back(x);
			dis[x][y] = z;
			dis[y][x] = z;
		}else{
			if (dis[x][y] > z){
				dis[x][y] = z;
				dis[y][x] = z;
			}
		}
	}
	prime();
	cout<<ans;
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0);
	cout.tie(0);
	ll t = 1;
	// cin>>t;
	while (t--)
	solve();
	return 0;
}
```

代码的复杂在于每次寻找最小边的时候都要进行两次循环先寻找点再寻找边，复杂度过高。



## 邻接矩阵优化：
```cpp
#include <bits/stdc++.h>
#define ll long long
using namespace std;
//n存储节点数量，m存储边的数量，ans是答案，cnt代表进入最小生成树节点的数量
int n,m,ans = 0, cnt = 0;
const int INF = 0x3f3f3f3f;
//边的最大值
const int N = 5010;
//dis[u][v]代表从u到v的权值，lowcost数组存储的是lowcost[u]代表u节点到最小生成树的距离（权值）
int dis[N][N],lowcost[N];
//via数组存储的是节点是否进入最小生成树
bool via[N];
void prim(){
    //1节点进入最小生成树
	via[1] = true;
    //进入最小生成树的节点+1
	cnt++;
    //初始化从1到各节点的距离（维护lowcost数组，或者说初始化与1这个最小生成树相连通的节点的距离）
	for (int i = 1;i<=n;++i)lowcost[i] = dis[1][i];
    //当节点数量为n时，最小生成树的边的数量为n-1，外层循环循环n-1次
	for (int k=1;k<n;++k){
        //minn变量存储的是本次循环得到的与最小生成树最短距离，u变量存储的是进入最小生成树的节点
		int minn = INF,u = -1;
        //循环各个节点
		for (int i=1;i<=n;++i){
            //判断节点i是否进入最小生成树，若没有并且节点i距离最小生成树的距离小于目前的minn，则i为新进入最小生成树的节点
			if (!via[i] && lowcost[i] < minn){
				minn = lowcost[i];
				u = i;
			}
		}
        //判断本次循环是否有节点未进入最小生成树，u=-1代表没有，但是倘若外层循环不结束，那么必定会有边进入最小生成树，如果没有则代表图中有节点不连通
		if (u == -1)break;
		via[u] = true;
		cnt++;
        //维护lowcost数组
		for (int i=1;i<=n;++i){
			if (!via[i] && lowcost[i] > dis[u][i]){
				lowcost[i] = dis[u][i];
			}
		}
		ans+=minn;
	}
}
void solve() {
	cin>>n>>m;
    //在节点未进入最小生成树前，所有节点的距离都初始化为0x3f
	memset(dis,0x3f,sizeof(dis));
    //从i节点到i节点的权值为0
	for (int i=1;i<n;++i)dis[i][i]=0;
    //读入数据
	for (int i=0;i<m;++i){
		int x,y,z;
		cin>>x>>y>>z;
		if (z<dis[x][y]){
			dis[x][y]=z;
			dis[y][x]=z;
		}
	}
	prim();
	if (cnt == n){
		cout<<ans;
	}else{
		cout<<"orz";
	}
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0);
	cout.tie(0);
	ll t = 1;
	// cin>>t;
	while (t--)
	solve();
	return 0;
}
```

最终得分：

![](资源归档/assets/91.png)

复杂度：`O(n^2)`

## Prim算法的最终形态，优先队列priority_queue（最小堆）优化
优先队列详见[优先队列](学习笔记/Algorithm/数据结构/优先队列（priority_queue）)

代码：

```cpp
#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int INF = 0x3f3f3f3f;
const int MAX_M = 1e5+5;
//边的最大值
const int N = 5010;
//n存储节点数量，m存储边的数量，ans是答案，cnt代表进入最小生成树节点的数量
int n,m,ans = 0,cnt = 0;
//图的邻接表
vector<vector<pair<int,int>>> graph;
//via数组代表对应节点是否进入最小生成树
bool via[N];
void prim(){
	//创建优先队列pd，greater<>最小堆
	priority_queue<pair<int,int>,vector<pair<int,int>>,greater<>> pd;
	//维护各个节点到生成树的距离的数组
	vector<int> dis(n+1,INF);
	dis[1] = 0;
	//将第一个节点进入生成树，权值为0
	pd.emplace(0,1);
	//如果优先队列为空，则代表该图的所有节点已经遍历完（可能存在其他图未连通）。如果cnt为n则代表所有节点已经进入最小生成树
	while(!pd.empty() && cnt<n){
		//访问优先队列中权值元素的最小值
		auto [w,u] = pd.top();
		//弹出最小的权值节点
		pd.pop();
		//倘若最小权值的节点已经进入最小生成树，则再重新循环
		if (via[u])continue;
		//倘若没有进入最小生成树则标记进入最小生成树
		via[u] = true;
		//增加答案
		ans+=w;
		//增加进入最小生成树的节点数量
		++cnt;
		for (auto [v,weight]:graph[u]){
			//v代表与u相连接的节点名，weight则代表权值
			//维护dis数组
			if(!via[v] && dis[v] > weight){
				dis[v] = weight;
				//倘若与u相连接的节点未进入最小生成树，则加入最小生成树
				pd.emplace(weight,v);
			}
		}
	}
}
void solve() {
	cin>>n>>m;
	vector<vector<int>> dis(n+1,vector<int> (n+1,INF));
	//读入数据
	for (int i=0;i<m;++i){
		int x,y,z;
		cin>>x>>y>>z;
		if (z<dis[x][y]){
			dis[x][y] = z;
			dis[y][x] = z;
		}
	}
	graph.resize(n+1);
	//选择平行边的最小值加入到邻接表中
	for (int i=1;i<=n;++i){
		for (int j=1;j<=n;++j){
			if (dis[i][j] != INF){
				graph[i].emplace_back(j,dis[i][j]);
				graph[j].emplace_back(i,dis[i][j]);
			}
		}
	}
	prim();
	if (cnt==n)cout<<ans;
	else cout<<"orz";
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0);
	cout.tie(0);
	ll t = 1;
	// cin>>t;
	while (t--)
	solve();
	return 0;
}
```

最终得分：

![](资源归档/assets/91.png)

复杂度：`O(m*log m)`

由于使用了优先队列优化，所以插入的复杂度降到了$\log n$，

时间320ms，Prim算法的最优解。

# Kruskal算法（克鲁斯卡尔算法）
## 定义：
普利姆（Prime）算法是以 点 为基础，挑选与 已有点 相连的最小边。

克鲁斯卡尔（Kruskal）算法是以 边 为基础，先将边从小到大排列，从小到大的添加不构成「环路」的边。而判断两条边有没有构成环路就需要使用一个知识：并查集，详见：[并查集](学习笔记/Algorithm/图论/并查集（DSU）)

并查集如何判断有没有环路呢，只需要查找两个点是否属于同一颗树下即可（即最小生成树）。

## 代码：
```cpp
#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 5010;
int n,m,ans = 0;
struct Edge{int u,v,w;};
//该算法采用边列表的形式存储
vector<Edge> graph;
//快排的依据，边权小的排在前面
bool cmp(Edge x,Edge y){
    return x.w<y.w;
}
//并查集，此处不在赘述
class DSU{
private:
    vector<int> parent;
    vector<int> rank;
    int count;
public:
    DSU(int n):count(n-1){
        parent.resize(n);
        rank.resize(n,0);
        for (int i=0;i<n;++i){
            parent[i] = i;
        }
    }
    int find(int x){
        if (parent[x] != x){
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }
    void join(int x,int y){
        int rootx = find(x),rooty = find(y);
        if (rootx == rooty)return;
        if (rank[rootx] < rank[rooty]){
            parent[rootx] = rooty;
        }else{
            parent[rooty] = rootx;
            if (rank[rootx] == rank[rooty])rank[rootx]++;
        }
        count--;
    }
    int getCount(){
        return count;
    }
};
void solve() {
    cin>>n>>m;
    //初始化并查集
    DSU d(n+1);
    //初始化边列表
    graph.resize(m);
    //读入数据
    for (int i=0;i<m;++i){
        cin>>graph[i].u>>graph[i].v>>graph[i].w;
    }
    //快排
    sort(graph.begin(),graph.end(),cmp);
    for (int i=0;i<graph.size();++i){
        //如果两个点都不在同一集合内（即不在最小生成树上，两个点连接在最小生成树上不构成环）
        if (d.find(graph[i].u) != d.find(graph[i].v)){
            //增加边权
            ans+=graph[i].w;
            //将两点加入最小生成树中
            d.join(graph[i].u,graph[i].v);
        }
    }
    //倘若只有一棵连通树，即所有点都包含在最小生成树中，输出ans
    if (d.getCount()==1){
        cout<<ans;
    }else{
        cout<<"orz";
    }
}
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    ll t = 1;
    // cin>>t;
    while (t--)
    solve();
    return 0;
}
```
