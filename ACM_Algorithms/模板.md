# 线段树，无MOD
```cpp
template <typename T>
class RangeTree {
    vector<T> tree, add_lazy, mul_lazy;
    int n, root, end, length;
    void build(int s, int t, int p, const vector<T>& a) {
        if (s == t) {
            tree[p] = a[s];
            return;
        }
        int mid = ((s + t) >> 1);
        build(s, mid, p * 2, a);
        build(mid + 1, t, p * 2 + 1, a);
        tree[p] = tree[p * 2] + tree[p * 2 + 1];
    }
    void maintain(int l, int r, int p) {
        if (mul_lazy[p] == 1 && add_lazy[p] == 0 || l == r) return;
        int mid = ((l + r) >> 1), left = p * 2, right = p * 2 + 1;
        tree[left] = tree[left] * mul_lazy[p] + add_lazy[p] * (mid - l + 1), mul_lazy[left] *= mul_lazy[p], add_lazy[left] = add_lazy[left] * mul_lazy[p] + add_lazy[p];
        tree[right] = tree[right] * mul_lazy[p] + add_lazy[p] * (r - mid), mul_lazy[right] *= mul_lazy[p], add_lazy[right] = add_lazy[right] * mul_lazy[p] + add_lazy[p];
        mul_lazy[p] = 1, add_lazy[p] = 0;
    }
    T range_query(int l, int r, int s, int t, int p) {
        if (l <= s && t <= r) return tree[p];
        int mid = ((s + t) >> 1), sum = 0;
        maintain(s, t, p);
        if (l <= mid) sum += range_query(l, r, s, mid, p * 2);
        if (r > mid) sum += range_query(l, r, mid + 1, t, p * 2 + 1);
        return sum;
    }
    void update_add(int l, int r, int s, int t, int p, T val) {
        if (l <= s && t <= r) {
            tree[p] += (t - s + 1) * val, add_lazy[p] += val;
            return;
        }
        int mid = ((s + t) >> 1);
        maintain(s, t, p);
        if (l <= mid) update_add(l, r, s, mid, p * 2, val);
        if (r > mid) update_add(l, r, mid + 1, t, p * 2 + 1, val);
        tree[p] = tree[p * 2] + tree[p * 2 + 1];
    }
    void update_mul(int l, int r, int s, int t, int p, T val) {
        if (l <= s && t <= r){
            tree[p] *= val, mul_lazy[p] *= val, add_lazy[p] *= val;
            return;
        }
        int mid = ((s + t) >> 1);
        maintain(s, t, p);
        if (l <= mid) update_mul(l, r, s, mid, p * 2, val);
        if (r > mid) update_mul(l, r, mid + 1, t, p * 2 + 1, val);
        tree[p] = tree[p * 2] + tree[p * 2 + 1];
    }
public:
    explicit RangeTree<T>(vector<T>& a) {
        n = a.size();
        length = n * 4;
        tree = vector<T>(length, 0);
        add_lazy = vector<T>(length, 0);
        mul_lazy = vector<T>(length, 1);
        end = n - 1;
        root = 1;
        build(0, end, 1, a);
    }
    void range_add(int l, int r, T val) { return update_add(l - 1, r - 1, 0, end, root, val); };
    void range_mul(int l, int r, T val) { return update_mul(l - 1, r - 1, 0, end, root, val); };
    T range_sum(int l, int r) { return range_query(l - 1, r - 1, 0, end, root); };
    void show(int p, int depth = 0) {
        if (p > length || tree[p] == 0) return;
        show(p * 2, depth + 1);
        for (int i = 0; i < depth; ++i) putchar('\t');
        printf("%d:%d\n", tree[p], add_lazy[p]);
        show(p * 2 + 1, depth + 1);
    }
};
```

```cpp
const int N = 5e5 + 5;

int n, m;
int bit[N];
inline int lowbit(int x) { return x & (-x); }
int query(int x) {
	int ans = 0;
	while (x) {
		ans += bit[x];
		x = x - lowbit(x);
	}
	return ans;
}
void add(int p, int k) {
	while (p <= n) {
		bit[p] += k;
		p = p + lowbit(p);
	}
}

	ios::sync_with_stdio(false);

		int x,j;

		j = i + lowbit(i);
		if (j <= n) bit[j] += bit[i];
	}

		int flag, x, y;
		cin >> flag >> x >> y;
		if (flag & 1) add(x, y);
```

# 欧拉函数
当我们需要求解$[1,n]$范围内所有数的欧拉函数，一个个计算时间复杂度过高，所以需要欧拉筛来辅助。（欧拉筛见[[素数筛法#欧拉筛（线性筛）| 欧拉筛]]）
我们已知欧拉筛在筛的过程中：
- 如果该数为质数，$\varphi(i) = i - 1$。
- 如果该数为合数，每个合数都会被它最小的质因数筛掉。设$p_j$为$x$的最小质因子，那么$x$就是被$p_j \times i$筛掉的，也就是$x = p_j \times i$。
	- 如果$p_j$是$i$的因数，那么$i$中已经包含了所有$x$的质因子。那么$${\Large
\begin{align}
\varphi(n) =& n \prod_{i=1}^{k} \frac{p_i - 1}{p_i} \\
=& p_j \times i \times \prod_{i=1}^{k} \frac{p_i - 1}{p_i} \\
=& p_j \times \varphi(i)
\end{align}
}$$
> [!tip]- 证明
> 我们$x$可以转化为$p_1^{\alpha_1} \times p_2^{\alpha_2} \times p_3^{\alpha_3} \times \dots \times p_k^{\alpha_k}$，由于$p_j$为$x$的最小质因子也就是式子中的$p_1$，而我们$x = p_j \times i = p_1 \times i$并且$i \% p_j = 0$所以$i = p_1^{\alpha_1 - 1} \times p_2^{\alpha_2} \times p_3^{\alpha_3} \times \dots \times p_k^{\alpha_k}$，即包含了所有$x$的质因子。


	- 如果$p_j$与$i$互质，那么就简单了$\varphi(x)=\varphi(p_j\times i)=\varphi(p_j) \times \varphi(i)=(p_j - 1) \times \varphi(i)$。

```cpp
int n,ans = 0;
cin>>n;
vector<int> oula(n + 1),fac;
vector<bool> flag(n + 1);
oula[1] = 1;
for (int i=2;i<=n;++i) {

    }

        int x = i * j;

        flag[x] = true;

            oula[x] = j * oula[i];
            break;
        }else oula[x] = (j - 1) * oula[i]; // p_j 与 i 互质
    }
}
```

# 树上LCA（倍增）
```cpp
#include <bits/stdc++.h>
#define int long long
#define ull unsigned long long
using namespace std;
const int INF = 1e18+2;
const int N = 1e5+5;
const int MOD = 1e9+7;
// int countbinary(int v) {return 63-__countl_zero(v);}
void solve() {
    int n,m,s;
    cin>>n>>m>>s;
    vector<int> h(n+1,0);
    vector<vector<int>> tree(n+1),fa(n+1,vector<int>(31));
    for (int i=1;i<n;++i){
        int u,v;
        cin>>u>>v;
        tree[u].emplace_back(v);
        tree[v].emplace_back(u);
    }
    h[s] = 1;
    function<void(int,int)> dfs = [&](int p,int father) -> void{
        h[p] = h[father] + 1;
        fa[p][0] = father;
        for (int i = 1;(1LL << i) <= h[p];++i) fa[p][i] = fa[fa[p][i-1]][i-1];
        for (const auto& i : tree[p]){
            if (h[i] == 0) dfs(i,p);
        }
    };
    function<int(int,int)> lca = [&](int x,int y)->int{
        if (h[x] < h[y]) swap(x,y);
        if (h[x] != h[y]) {
            int dh = h[x] - h[y];
            for (int i=0;i<31 && (1LL << i) <= dh;++i){
                int _ = (1LL << i);
                if (dh & _) x = fa[x][i];
            }
        }
        if (x == y) return x;
        for (int i = 30;i>=0;--i){
            if (h[fa[x][i]] == 0) continue;
            if (fa[x][i] == fa[y][i]) continue;
            else{
                x = fa[x][i];
                y = fa[y][i];
            }
        }
        return fa[x][0];
    };
    dfs(s,0);
    while(m--){
        int u,v;
        cin>>u>>v;
        cout<<lca(u,v)<<'\n';
    }
}
signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    // cout << fixed << setprecision(2);
    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```

# DSU ON TREE
题目描述
给一棵根为1的树，每次询问子树颜色种类数
输入格式
第一行一个整数n，表示树的结点数
接下来n-1行，每行一条边
接下来一行n个数，表示每个结点的颜色`c[i]`
接下来一个数m，表示询问数
接下来m行表示询问的子树
输出格式
对于每个询问，输出该子树颜色数
输入输出样例 #1

输入 #1

```plain
5
1 2
1 3
2 4
2 5
1 2 2 3 3
5
1
2
3
4
5
```

输出 #1

```plain
3
2
1
1
1
```

说明/提示

对于前三组数据，有 $1\leq m,c[i]\leq n\leq 100$

而对于所有数据，有$1\leq m,c[i]\leq n\leq 1e5$

由并查集的启发式算法进行推广，朴素的树上合并通常需要我们在对应每个节点的子树进行遍历然后再合并，时间复杂度为$O(n^{2})$，而题目中又是离线的，为了尽可能优化时间复杂度，贪心的想，我们在遍历的时候倘若优先处理节点较多的子树，然后在下一次遍历的时候将节点较少的子树合并到节点较大的树上就可以优化时间复杂度，重点就在于：**将轻子树合并到重子树上**。时间复杂度为$O(n\log{n})$

证明需要用到树链剖分的知识，这部分还没学习，等学习完了再来。

优化
前面提到，DSU on Tree利用轻重子树概念加速了合并的效率。那么我们可以直接利用在重链剖分的时候得到dfs序，由递归转化为迭代，优化合并的常数。

Code
```cpp
#include <bits/stdc++.h>
#define int long long
#define ull unsigned long long
using namespace std;
const int N = 1e5+5;
// n为节点数，m为询问数量，totdfs为dfs序的计数器，totcol为颜色数量的计数器
int n,m,totdfs,totcol;
// grid为题目中构建的树
vector<int> grid[N];
/*cnt数组为对应颜色k的数量，Node数组记录的是对应dfs序的节点，
L为以这个节点为根的树的dfs序的开始，R为结束
son数组记录的是该节点重子树的根节点，col为对应i节点的颜色，
sz为该节点树的大小，ans数组为对应i节点颜色的数量即为答案
*/
int cnt[N],Node[N],L[N],R[N],son[N],col[N],sz[N],ans[N];

// 将节点加入到答案中
void add(int p){
    if (cnt[col[p]] == 0) totcol++;
    cnt[col[p]]++;
}
// 从答案中删除
void del(int p){
    cnt[col[p]]--;
    if (cnt[col[p]] == 0) --totcol;
}
// 第一次遍历，确定dfs序和重子节点
void dfs0(int u,int p){
    sz[u] = 1;
    L[u] = ++totdfs;
    Node[totdfs] = u;
    for (const auto& i:grid[u]){
        if (i != p){
            dfs0(i,u);
            sz[u] += sz[i];
            if (sz[son[u]] < sz[i]) son[u] = i;
        }
    }
    R[u] = totdfs;
}
// 第二次遍历进行DSU on Tree，方法中u为当前节点，p为当前节点的父节点，
// keep为是否保留，即是否为轻重子树
void dfs1(int u,int p,bool keep){
    // 轻子树不保留，只计算答案
    for (const auto&i:grid[u]){
        if (i != p && i != son[u]) dfs1(i,u,false);
    }
    // 计算重子树，并保留答案进入cnt中
    if (son[u]) dfs1(son[u],u,true);
    // 将轻子树的颜色添加到重子树中去
    for (const auto&i:grid[u]){
        if (i == p || i == son[u]) continue;
        for (int j = L[i];j<=R[i];++j) add(Node[j]);
    }
    // 添加当前节点的颜色，并计算答案
    add(u);
    ans[u] = totcol;
    // 倘若为轻子树则删除节点的颜色在cnt中的数量
    if (!keep) {
        for (int i=L[u];i<=R[u];++i) del(Node[i]);
    }
}
void solve() {
    cin>>n;
    for (int i=1;i<=n-1;++i){
        int u,v;
        cin>>u>>v;
        grid[u].emplace_back(v);
        grid[v].emplace_back(u);
    }
    for (int i=1;i<=n;++i) cin>>col[i];
    cin>>m;
    dfs0(1,0);
    dfs1(1,0,false);
    while (m--){
        int k;
        cin>>k;
        cout<<ans[k]<<'\n';
    }
}
signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    // cout << fixed << setprecision(2);
    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```

# 高斯消元
```cpp
#include <bits/stdc++.h>
#define ull unsigned long long
#define i128 __int128_t
using namespace std;
const int INF = 1e18;
const int N = 105;
const int MOD = 1e9 + 7;
const double eqs = 1e-8;
int n;
double matrix[N][N], ans[N];
int main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);
	cout << fixed << setprecision(4);
	cin >> n;
	for (int i = 1; i <= n; ++i) 
		for (int j = 1; j <= n + 1; ++j) 
			cin >> matrix[i][j];
	int r = 1; // 在这里，我们去寻找每一列中的主元
	for (int c = 1; c <= n; ++c) {
		int p = r;
		// 去寻找尽量大的数当作主元
		for (int i = r; i <= n; ++i) 
			if (fabs(matrix[p][c]) < fabs(matrix[i][c])) p = i;
		// 当前列没有有效的主元（即全为零）
		// 那么就跳过这一列的主元（不跳过行，所以需要另外再创建一个r的参数作为寻找过的行）
		if (fabs(matrix[p][c]) <= eqs) continue;
		// 设置主元
		swap(matrix[r], matrix[p]);
		// 消去当前列下方的所有元素
		for (int i = r + 1; i <= n; ++i) {
			double fac = matrix[i][c] / matrix[r][c];
			for (int j = 1; j <= n + 1; ++j) matrix[i][j] -= fac * matrix[r][j];
		}
		r++; // 非零行加1
	}
	for (int i = r; i <= n; ++i) {
		bool flag = true;
		for (int j = 1; j <= n; ++j) {
			if (fabs(matrix[i][j]) > eqs) {
				flag = false;
				break;
			}
		}
		// 系数全零，但是右端项非零，形成了0 = c的式子，所以无解
		if (flag && fabs(matrix[i][n + 1]) > eqs) {
			cout << -1 << '\n';
			return 0;
		}
	}
	// 倘若存在0 = 0的式子，那么就存在无穷多解
	if (r <= n) {
		cout << 0 << '\n';
		return 0;
	}
	for (int i = n; i >= 1; --i) {
		// 开始消元
		for (int j = i + 1; j <= n; ++j) matrix[i][n + 1] -= ans[j] * matrix[i][j];
		// 求解出对应x
		ans[i] = matrix[i][n + 1] / matrix[i][i];
	}
	for (int i = 1; i <= n; ++i) cout << 'x' << i << '=' << ans[i] << '\n';
	return 0;
}
```

# 线性基
```cpp
struct XorLinearBasis {
private:
	int rank;
	vector<ll> basis;
public:
	XorLinearBasis() {
		basis.resize(LOG);
		rank = 0;
	}
	// 插入并维护基
	void insert(ll x) {
		for (int i = LOG - 1; i >= 0; --i) {
			if (!((x >> i) & 1)) continue;
			if (!basis[i]) { // 高斯消元
				basis[i] = x;
				rank++;
			}
			x ^= basis[i];
		}
	}
	// 正交化，就是将所有化成 1
	void normalize() {
		for (int i = LOG - 1; i >= 0; --i) {
			if (!basis[i]) continue;
			for (int j = i - 1; j >= 0; --j) 
				if ((basis[i] >> j) & 1) 
					basis[i] ^= basis[j];
		}
	}
	// 获取最大异或和
	ll max_xor() {
		ll res = 0;
		for (int i = LOG - 1; i >= 0; --i) res = max(res, res ^ basis[i]);
		return res;
	}
	// 获取最小异或和
	ll min_xor() {
		for (int i = 0; i < LOG; ++i) if (basis[i]) return basis[i];
		return 0;
	}
	// 判断这个数字x能否被这个线性基表示
	bool is_representable(ll x) {
		for (int i = LOG - 1; i >= 0; --i) {
			if (!((x >> i) & 1)) continue;
			if (!basis[i]) return false;
			x ^= basis[i];
		}
		return true;
	}
	// 获取第 k 小值
	ll k_xor(ll k) {
		normal_distribution();
		if (k == 1) return 0;
		k--;
		ll res = 0;
		vector<ll> v;
		for (int i = LOG - 1; i >= 0; --i) if (basis[i]) v.push_back(basis[i]);
		while (!v.empty() && k) {
			if (k & 1) res ^= v.back();
			v.pop_back();
			k >>= 1;
		}
		if (k) return -1;
		else return res;
	}
};
```

# 欧拉筛（线性筛）
埃氏筛法仍有优化空间，它会将一个合数重复多次标记．有没有什么办法省掉无意义的步骤呢？答案是肯定的。
我们的做法是，让每⼀个合数被它的最⼩质因数筛掉。
如果能让每个合数都只被标记一次，那么时间复杂度就可以降到 $O(n)$ 了。
```cpp
const int N = 1e6 + 5;
vector<int> fac;
vector<bool> flag(N);
void init() {
    flag[1] = true; // 如果没标记过，就是质数
    for (int i=2;i<N;++i) {
        if (!flag[i]) fac.push_back(i);
        for (const int& j : fac) {
            if (i * j >= N) break;
            flag[i * j] = true;
            if (i % j == 0) break;
            // i % j == 0，换言之，i 之前被 j 筛过了，也就是说 j 是 i 的最小质因子，由于 fac 里面质数是从小到大的，所以 i 乘上其他的质数的结果一定会被 j 的倍数筛掉，就不需要在这里先筛一次，所以这里直接 break 掉就好了。
            // 例如，当 i 为 4 时，fac 数组里面有[2,3]，如果不直接在 i 为 2 的时候 break 掉，那么对于 12 这个数字来说，最小质因子应该是 2，在 4 的时候会被筛掉，复杂度就不是O(n)了。
        }
    }
}
```

# 莫队
题目大意：有一个长度为 $n$ 的序列 $\{c_i\}$。现在给出 $m$ 个询问，每次给出两个数 $l,r$，从编号在 $l$ 到 $r$ 之间的数中随机选出两个不同的数，求两个数相等的概率。
```cpp
// BLOCK为块的大小
int n,m,BLOCK;
struct Querry {
	int l,r,id;
	bool operator<(const node &x) const {
	    if (l / unit != x.l / unit) return l < x.l;
	    // 注意下面两行不能写小于（大于）等于，否则会出错
	    if ((l / unit) & 1) return r < x.r;
	    return r > x.r;
	}
}q[N];
int sum;
int c[N],ans1[N],ans2[N],col[N];
// 添加颜色的时候
void add(int i) {
	// 先加上2col[i]
	sum += 2 * col[i];
	// 再更新
	col[i]++;
}
void del(int i) {
	// 先更新
	col[i]--;
	// 再减去2col[i]
	sum -= 2 * col[i];
}
signed main() {
	cin>>n>>m;
	BLOCK = sqrt(n);
	for (int i = 1;i <= n;++i) cin>>c[i];
	// 对每个查询分别输入
	for (int i = 1;i <= m;++i) {
		cin>>q[i].l>>q[i].r;
		q[i].id = i;
	}
	// 对查询排序
	sort(q + 1,q + 1 + m);
	for (int i = 1,l = 1,r = 0;i <= m;++i) {
		// 在一个点内的时候，是肯定不可能存在有效的解的，直接输出概率为0的情况
		if (q[i].l == q[i].r) {
			ans1[q[i].id] = 0;
			ans2[q[i].id] = 1;
			continue;
		}
		// 这里四个区间更新的顺序千万不能乱，原因下面会提到
		// 左区间需要往外扩张，那么就先移动l，然后加上c[l]的值
		while(l > q[i].l) add(c[--l]);
		// 右区间需要往外扩张，那么就先移动r，然后加上c[r]的值
		while(r < q[i].r) add(c[++r]);
		// 左区间范围太大，需要缩小，那么就先减去c[l]的值，再移动l
		while(l < q[i].l) del(c[l++]);
		// 右区间范围太大，需要缩小，那么就先减去c[r]的值，再移动r
		while(r > q[i].r) del(c[r--]);
		int res1 = sum,res2 = (q[i].r - q[i].l + 1) * (q[i].r - q[i].l);
		// 约分操作
		int _ = __gcd(res1,res2);
		res1 /= _;
		res2 /= _;
		ans1[q[i].id] = res1;
		ans2[q[i].id] = res2;
	}
	for (int i = 1;i <= m;++i) cout<<ans1[i]<<'/'<<ans2[i]<<'\n';
	return 0;
}
```
